# SLLM Probe - 함대 사령부(제독 전투 기록·청문회, 2026-10-17). 모드의 on_actions 는 게임 것과 합쳐진다. 숨은 이벤트만 부른다(4.5.2: on_actions 에 effect 블록 불가).
# 범위는 바닐라 00_on_actions.txt 주석: This = 함대 1 소유국, From = 함대 2 소유국, FromFrom = 함대 1, FromFromFrom = 함대 2.
on_space_battle_won = {
	events = { sllm_fleet.1 }
}

	events = { sllm_fleet.2 }
}

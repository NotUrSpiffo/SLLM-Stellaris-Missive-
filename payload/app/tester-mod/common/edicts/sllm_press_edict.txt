    icon = "GFX_edict_type_policy"

        cost = { unity = 0 }

    potential = { is_ai = no is_gestalt = no } # 단일 의식(기계 지성·군체)에는 자유 언론이 없다(2026-10-19, 바닐라 scripted trigger is_gestalt)

        sllm_probe_press_prepare = yes
        custom_tooltip = sllm_press_edict_start_tt
    }

    on_disabled = {
        custom_tooltip = sllm_press_edict_off_tt
    }

    ai_weight = { weight = 0 }

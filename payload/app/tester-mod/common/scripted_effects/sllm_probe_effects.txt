# Test 9: give every regular empire our own index number and log it.
# The app reads SLLM:IDX lines to map names to numbers. Scope: player country.
sllm_probe_assign_idx = {
	if = { limit = { NOT = { is_variable_set = sllm_idx_next } } set_variable = { which = sllm_idx_next value = 1 } }
	# A country that inherited another country's variables (e.g. copy_flags_and_variables_from)
	# has the number but not the owner flag that carries its own ID. Clear and renumber it.
	every_country = {
		limit = {
			is_variable_set = sllm_idx
			NOT = { has_country_flag = sllm_own_@this }
		}
		log = "SLLM:IDX_COPY/;/name=\\[This.GetName]/;/old=\\[This.sllm_idx]"
		clear_variable = sllm_idx
	}
	every_country = {
		limit = {
			OR = {
				is_country_type = default
				is_country_type = fallen_empire
				is_country_type = awakened_fallen_empire
				has_communications = prev
			}
			NOT = { is_variable_set = sllm_idx }
		}
		set_variable = { which = sllm_idx value = prev.sllm_idx_next }
		set_country_flag = sllm_own_@this
		prev = { change_variable = { which = sllm_idx_next value = 1 } }
	}
	# Known powers, read live from the game instead of the save file.
	# rel = self | comm (communications open, can talk) | contact (met, no communications yet) | none
	# prev here is the human player country that ran this effect.
	# Checked from the other empire towards the player (communications are mutual).
	every_country = {
		limit = { is_variable_set = sllm_idx }
		if = {
			limit = { is_same_value = prev }
			log = "SLLM:IDX/;/name=\\[This.GetName]/;/idx=\\[This.sllm_idx]/;/type=\\[This.GetCountryType]/;/rel=self"
		}
		else_if = {
			limit = { has_communications = prev }
			log = "SLLM:IDX/;/name=\\[This.GetName]/;/idx=\\[This.sllm_idx]/;/type=\\[This.GetCountryType]/;/rel=comm"
		}
		else_if = {
			limit = { has_established_contact = prev }
			log = "SLLM:IDX/;/name=\\[This.GetName]/;/idx=\\[This.sllm_idx]/;/type=\\[This.GetCountryType]/;/rel=contact"
		}
		else = {
			log = "SLLM:IDX/;/name=\\[This.GetName]/;/idx=\\[This.sllm_idx]/;/type=\\[This.GetCountryType]/;/rel=none"
		}
	}
}

# Test 10: one call does everything inline, no event in between.
# Console: effect sllm_probe_apply = { REQ = 2001 TID = 2 AMT = 100 }
# Works from any selection because it scopes to the human player itself.
sllm_probe_apply = {
	every_playable_country = {
		limit = { is_ai = no }
		sllm_probe_init_vars = yes
		set_variable = { which = sllm_in_req value = $REQ$ }
		if = {
			limit = { check_variable = { which = sllm_last_req value < $REQ$ } }
			set_variable = { which = sllm_last_req value = $REQ$ }
			change_variable = { which = sllm_exec_seq value = 1 }
			if = {
				limit = {
					any_country = {
						is_variable_set = sllm_idx
						check_variable = { which = sllm_idx value = $TID$ }
						has_resource = { type = energy amount >= $AMT$ }
					}
				}
				random_country = {
					limit = {
						is_variable_set = sllm_idx
						check_variable = { which = sllm_idx value = $TID$ }
					}
					add_resource = { energy = -$AMT$ }
					log = "SLLM:ACK_TARGET/;/\\[This.GetName]"
				}
				add_resource = { energy = $AMT$ }
				set_variable = { which = sllm_last_result value = 1 }
				log = "SLLM:ACK/;/seq=\\[This.sllm_exec_seq]/;/req=\\[This.sllm_in_req]/;/apply/;/ok"
			}
			else = {
				set_variable = { which = sllm_last_result value = 2 }
				log = "SLLM:ACK/;/seq=\\[This.sllm_exec_seq]/;/req=\\[This.sllm_in_req]/;/apply/;/fail/;/no_target_or_energy"
			}
		}
		# Same id again: repeat the stored outcome, change nothing.
		else_if = {
			limit = { check_variable = { which = sllm_last_req value = $REQ$ } }
			log = "SLLM:ACK/;/seq=\\[This.sllm_exec_seq]/;/req=\\[This.sllm_in_req]/;/apply/;/dup/;/result=\\[This.sllm_last_result]"
		}
		# Older id: a late or replayed request. Refuse.
		else = {
			log = "SLLM:ACK/;/seq=\\[This.sllm_exec_seq]/;/req=\\[This.sllm_in_req]/;/apply/;/stale/;/last=\\[This.sllm_last_req]"
		}
	}
}


# 상대 제국의 자원 사정(2026-10-17): 지도자가 "요즘 합금이 부족합니다" 같은 말을 하게 하는 근거.
# 상태 값: 0 보통, 1 재고 부족, 2 월 적자, 3 재고 부족 + 월 적자(심각), 4 풍족(재고 많고 월 흑자).
# 범위: 대상 제국(AI). 값은 PROFILE 줄의 ec= 에 에너지·광물·식량·합금·소비재 순으로 한 자리씩 붙인다.
sllm_probe_econ_one = {
	set_variable = { which = $VAR$ value = 0 }
	if = {
		limit = { has_resource = { type = $RES$ amount >= $HI$ } has_monthly_income = { resource = $RES$ value > 0 } }
		set_variable = { which = $VAR$ value = 4 }
	}
	if = {
		limit = { has_monthly_income = { resource = $RES$ value < 0 } }
		set_variable = { which = $VAR$ value = 2 }
	}
	if = {
		limit = { has_resource = { type = $RES$ amount < $LO$ } }
		if = {
			limit = { check_variable = { which = $VAR$ value = 2 } }
			set_variable = { which = $VAR$ value = 3 }
		}
		else = { set_variable = { which = $VAR$ value = 1 } }
	}
}
sllm_probe_econ_state = {
	sllm_probe_econ_one = { VAR = sllm_ec_e RES = energy LO = 100 HI = 5000 }
	sllm_probe_econ_one = { VAR = sllm_ec_m RES = minerals LO = 100 HI = 5000 }
	sllm_probe_econ_one = { VAR = sllm_ec_f RES = food LO = 50 HI = 3000 }
	sllm_probe_econ_one = { VAR = sllm_ec_a RES = alloys LO = 40 HI = 2000 }
	sllm_probe_econ_one = { VAR = sllm_ec_c RES = consumer_goods LO = 30 HI = 1500 }
}

# Energy transfer with a direction. Console/app: effect every_playable_country = { limit = { is_ai = no } sllm_probe_transfer = { REQ = 5 TID = 2 AMT = 50 DIR = 1 } }
# DIR 1 = the AI empire gives energy to the player. DIR 2 = the player gives energy to the AI empire.
# Result variable: sllm_last_result 1 = done, 2 = failed (no such empire or not enough energy).
sllm_probe_transfer = {
	every_playable_country = {
		limit = { is_ai = no }
		sllm_probe_init_vars = yes
		set_variable = { which = sllm_in_req value = $REQ$ }
		set_variable = { which = sllm_in_dir value = $DIR$ }
		if = {
			limit = { check_variable = { which = sllm_last_req value < $REQ$ } }
			set_variable = { which = sllm_last_req value = $REQ$ }
			change_variable = { which = sllm_exec_seq value = 1 }
			set_variable = { which = sllm_last_result value = 2 }
			if = {
				limit = {
					check_variable = { which = sllm_in_dir value = 1 }
					any_country = {
						is_variable_set = sllm_idx
						check_variable = { which = sllm_idx value = $TID$ }
						has_resource = { type = energy amount >= $AMT$ }
					}
				}
				random_country = {
					limit = {
						is_variable_set = sllm_idx
						check_variable = { which = sllm_idx value = $TID$ }
					}
					add_resource = { energy = -$AMT$ }
				}
				add_resource = { energy = $AMT$ }
				set_variable = { which = sllm_last_result value = 1 }
			}
			else_if = {
				limit = {
					check_variable = { which = sllm_in_dir value = 2 }
					has_resource = { type = energy amount >= $AMT$ }
					any_country = {
						is_variable_set = sllm_idx
						check_variable = { which = sllm_idx value = $TID$ }
					}
				}
				random_country = {
					limit = {
						is_variable_set = sllm_idx
						check_variable = { which = sllm_idx value = $TID$ }
					}
					add_resource = { energy = $AMT$ }
				}
				add_resource = { energy = -$AMT$ }
				set_variable = { which = sllm_last_result value = 1 }
			}
			log = "SLLM:ACK/;/seq=\\[This.sllm_exec_seq]/;/req=\\[This.sllm_in_req]/;/xfer/;/dir=\\[This.sllm_in_dir]/;/result=\\[This.sllm_last_result]"
		}
		else_if = {
			limit = { check_variable = { which = sllm_last_req value = $REQ$ } }
			log = "SLLM:ACK/;/seq=\\[This.sllm_exec_seq]/;/req=\\[This.sllm_in_req]/;/xfer/;/dup/;/result=\\[This.sllm_last_result]"
		}
		else = {
			log = "SLLM:ACK/;/seq=\\[This.sllm_exec_seq]/;/req=\\[This.sllm_in_req]/;/xfer/;/stale/;/last=\\[This.sllm_last_req]"
		}
	}
}

# Conversation mood: the AI empire TID gets an opinion modifier towards the player.
# STEP 1 warm small, 2 warm large, 3 cold small, 4 cold large.
# Logs SLLM:OPSHIFT (new opinion of the AI towards the player) and then SLLM:ACK with the same request id.
sllm_probe_opinion_shift = {
	every_playable_country = {
		limit = { is_ai = no }
		sllm_probe_init_vars = yes
		set_variable = { which = sllm_in_req value = $REQ$ }
		set_variable = { which = sllm_in_step value = $STEP$ }
		if = {
			limit = { check_variable = { which = sllm_last_req value < $REQ$ } }
			set_variable = { which = sllm_last_req value = $REQ$ }
			change_variable = { which = sllm_exec_seq value = 1 }
			set_variable = { which = sllm_last_result value = 2 }
			if = {
				limit = {
					check_variable = { which = sllm_in_step value > 0 }
					check_variable = { which = sllm_in_step value < 5 }
					any_country = {
						is_variable_set = sllm_idx
						check_variable = { which = sllm_idx value = $TID$ }
					}
				}
				random_country = {
					limit = {
						is_variable_set = sllm_idx
						check_variable = { which = sllm_idx value = $TID$ }
					}
					if = {
						limit = { prev = { check_variable = { which = sllm_in_step value = 1 } } }
						add_opinion_modifier = { modifier = sllm_talk_warm_small who = prev }
					}
					else_if = {
						limit = { prev = { check_variable = { which = sllm_in_step value = 2 } } }
						add_opinion_modifier = { modifier = sllm_talk_warm_large who = prev }
					}
					else_if = {
						limit = { prev = { check_variable = { which = sllm_in_step value = 3 } } }
						add_opinion_modifier = { modifier = sllm_talk_cold_small who = prev }
					}
					else = {
						add_opinion_modifier = { modifier = sllm_talk_cold_large who = prev }
					}
					sllm_probe_trust_by_step = yes
					export_trigger_value_to_variable = {
						trigger = opinion
						parameters = { who = prev }
						variable = sllm_shift_op
					}
					log = "SLLM:OPSHIFT/;/idx=\\[This.sllm_idx]/;/op=\\[This.sllm_shift_op]/;/req=\\[Prev.sllm_in_req]"
				}
				set_variable = { which = sllm_last_result value = 1 }
			}
			log = "SLLM:ACK/;/seq=\\[This.sllm_exec_seq]/;/req=\\[This.sllm_in_req]/;/shift/;/step=\\[This.sllm_in_step]/;/result=\\[This.sllm_last_result]"
		}
		else_if = {
			limit = { check_variable = { which = sllm_last_req value = $REQ$ } }
			log = "SLLM:ACK/;/seq=\\[This.sllm_exec_seq]/;/req=\\[This.sllm_in_req]/;/shift/;/dup/;/result=\\[This.sllm_last_result]"
		}
		else = {
			log = "SLLM:ACK/;/seq=\\[This.sllm_exec_seq]/;/req=\\[This.sllm_in_req]/;/shift/;/stale/;/last=\\[This.sllm_last_req]"
		}
	}
}

# SLLM Probe - 평화 중재·강제 평화 (2026-10-17). Stellaris 4.5.2 실제 동작은 게임 검수 필요.
# 모든 효과는 플레이어 국가 범위(This = 플레이어)에서 부른다. 로그 문자열에는 매개변수를 넣지 않는다(F-39).
#  전쟁 목록 : sllm_probe_med_wars = { REQ }                    → MEDWAR* , MEDWARS_END
#  점령 읽기 : sllm_probe_med_find = { REQ A D } (AI 끼리)  또는 sllm_probe_med_find_player = { REQ TID } (플레이어 전쟁)
#              뒤에 sllm_probe_med_scan = yes                     → MEDSYS* , MEDSCAN_END
#  적용      : find… 뒤에 sllm_probe_med_give = { FROM TO SID }* 와 sllm_probe_med_peace = yes → MEDGIVE* , MEDAPPLY_END
# 당사국은 event_target:sllm_med_p1(주 공격국 · 플레이어 전쟁이면 플레이어) / sllm_med_p2, 전쟁은 event_target:sllm_med_war.
# 성계 번호 sllm_med_sid 는 플레이어 국가의 카운터로 한 번만 붙이고 바꾸지 않는다(같은 성계는 같은 번호).

# 플레이어가 끼지 않은 일반 제국(is_country_type = default) 끼리의 전쟁(주 공격국·주 방어국 모두 통신·번호 있는 나라)
sllm_probe_med_wars = {
	set_variable = { which = sllm_med_req value = $REQ$ }
	save_event_target_as = sllm_med_viewer
	every_country = {
		limit = { is_ai = yes is_country_type = default is_variable_set = sllm_idx has_communications = prev is_at_war = yes }
		set_variable = { which = sllm_med_rq value = prev.sllm_med_req }
		set_country_flag = sllm_med_it
		every_war = {
			limit = {
				attacker = { has_country_flag = sllm_med_it }
				defender = { is_ai = yes is_country_type = default is_variable_set = sllm_idx has_communications = event_target:sllm_med_viewer }
				NOT = { any_war_participant = { is_ai = no } }
			}
			set_variable = { which = sllm_med_rq value = prev.sllm_med_rq }
			export_trigger_value_to_variable = { trigger = attacker_war_exhaustion variable = sllm_med_ax }
			export_trigger_value_to_variable = { trigger = defender_war_exhaustion variable = sllm_med_dx }
			attacker = { prev = { set_variable = { which = sllm_med_a value = prev.sllm_idx } } }
			defender = { prev = { set_variable = { which = sllm_med_d value = prev.sllm_idx } } }
			log = "SLLM:MEDWAR/;/req=\\[This.sllm_med_rq]/;/name=\\[This.GetName]/;/a=\\[This.sllm_med_a]/;/d=\\[This.sllm_med_d]/;/ax=\\[This.sllm_med_ax]/;/dx=\\[This.sllm_med_dx]"
		}
		remove_country_flag = sllm_med_it
	}
	log = "SLLM:MEDWARS_END/;/req=\\[This.sllm_med_req]/;/sid=\\[This.sllm_save_id]"
}

# AI 끼리의 전쟁 찾기: 주 공격국 번호 A, 주 방어국 번호 D, 플레이어 불참.
sllm_probe_med_find = {
	set_variable = { which = sllm_med_req value = $REQ$ }
	remove_country_flag = sllm_med_found
	save_event_target_as = sllm_med_player
	every_country = {
		limit = { is_ai = yes is_country_type = default is_variable_set = sllm_idx check_variable = { which = sllm_idx value = $A$ } has_communications = prev }
		set_country_flag = sllm_med_it
		every_war = {
			limit = {
				attacker = { has_country_flag = sllm_med_it }
				defender = { is_ai = yes is_country_type = default is_variable_set = sllm_idx check_variable = { which = sllm_idx value = $D$ } }
				NOT = { any_war_participant = { is_ai = no } }
			}
			save_event_target_as = sllm_med_war
			attacker = { save_event_target_as = sllm_med_p1 }
			defender = { save_event_target_as = sllm_med_p2 }
			event_target:sllm_med_player = { set_country_flag = sllm_med_found }
		}
		remove_country_flag = sllm_med_it
	}
	sllm_probe_med_prepare = yes
}

# 플레이어와 AI(번호 TID) 둘만의 전쟁 찾기(강제 평화). 다른 나라가 낀 전쟁은 다루지 않는다.
sllm_probe_med_find_player = {
	set_variable = { which = sllm_med_req value = $REQ$ }
	remove_country_flag = sllm_med_found
	save_event_target_as = sllm_med_player
	save_event_target_as = sllm_med_p1
	set_country_flag = sllm_med_it
	every_country = {
		limit = { is_ai = yes is_variable_set = sllm_idx check_variable = { which = sllm_idx value = $TID$ } is_at_war_with = prev }
		save_event_target_as = sllm_med_p2
		every_war = {
			limit = {
				any_war_participant = { has_country_flag = sllm_med_it }
				count_war_participant = { count = 2 }
			}
			save_event_target_as = sllm_med_war
			event_target:sllm_med_player = { set_country_flag = sllm_med_found }
		}
	}
	remove_country_flag = sllm_med_it
	sllm_probe_med_prepare = yes
}

sllm_probe_med_prepare = {
	set_variable = { which = sllm_med_ok value = 0 }
	if = {
		limit = { has_country_flag = sllm_med_found }
		set_variable = { which = sllm_med_ok value = 1 }
		event_target:sllm_med_p1 = { set_variable = { which = sllm_med_rq value = prev.sllm_med_req } }
		event_target:sllm_med_p2 = { set_variable = { which = sllm_med_rq value = prev.sllm_med_req } }
	}
}

# 성계(This)에 고정 번호를 붙인다. 플레이어 국가의 카운터를 쓴다.
sllm_probe_med_tag = {
	if = {
		limit = { NOT = { is_variable_set = sllm_med_sid } }
		event_target:sllm_med_player = {
			if = { limit = { NOT = { is_variable_set = sllm_med_sysn } } set_variable = { which = sllm_med_sysn value = 0 } }
			change_variable = { which = sllm_med_sysn value = 1 }
			prev = { set_variable = { which = sllm_med_sid value = prev.sllm_med_sysn } }
		}
	}
}

# OWN 이 가진 성계 가운데 OCC 가 점령한 곳(궤도 기지나 OWN 의 행성을 OCC 가 장악). SIDE 1 = p1 소유, 2 = p2 소유.
sllm_probe_med_occupied = {
	event_target:sllm_med_$OWN$ = {
		every_system_within_border = {
			limit = {
				OR = {
					starbase = { is_controlled_by = event_target:sllm_med_$OCC$ }
					any_system_planet = { is_owned_by = event_target:sllm_med_$OWN$ is_controlled_by = event_target:sllm_med_$OCC$ }
				}
			}
			sllm_probe_med_tag = yes
			set_variable = { which = sllm_med_rq value = prev.sllm_med_rq }
			set_variable = { which = sllm_med_side value = $SIDE$ }
			set_variable = { which = sllm_med_cl value = 0 }
			if = { limit = { event_target:sllm_med_$OCC$ = { has_claim = prev } } set_variable = { which = sllm_med_cl value = 1 } }
			set_variable = { which = sllm_med_sb value = 0 }
			if = { limit = { starbase = { is_controlled_by = event_target:sllm_med_$OCC$ } } set_variable = { which = sllm_med_sb value = 1 } }
			set_variable = { which = sllm_med_pl value = 0 }
			set_variable = { which = sllm_med_po value = 0 }
			every_system_planet_colony = {
				limit = { is_owned_by = event_target:sllm_med_$OWN$ }
				prev = { change_variable = { which = sllm_med_pl value = 1 } }
				if = { limit = { is_controlled_by = event_target:sllm_med_$OCC$ } prev = { change_variable = { which = sllm_med_po value = 1 } } }
			}
			log = "SLLM:MEDSYS/;/req=\\[This.sllm_med_rq]/;/side=\\[This.sllm_med_side]/;/sid=\\[This.sllm_med_sid]/;/name=\\[This.GetName]/;/claim=\\[This.sllm_med_cl]/;/sb=\\[This.sllm_med_sb]/;/pl=\\[This.sllm_med_pl]/;/po=\\[This.sllm_med_po]"
		}
	}
}

sllm_probe_med_scan = {
	if = {
		limit = { has_country_flag = sllm_med_found }
		sllm_probe_med_occupied = { OWN = p1 OCC = p2 SIDE = 1 }
		sllm_probe_med_occupied = { OWN = p2 OCC = p1 SIDE = 2 }
	}
	log = "SLLM:MEDSCAN_END/;/req=\\[This.sllm_med_req]/;/sid=\\[This.sllm_save_id]/;/found=\\[This.sllm_med_ok]"
	remove_country_flag = sllm_med_found
}

# 점령·영유권 조건을 적용 시점에 다시 확인하고, 맞으면 성계(궤도 기지)와 FROM 소유 행성을 TO 에게 넘긴다.
# hit 0 = 조건 불충족·성계 없음, 1 = 넘김(궤도 기지 소유 확인 못 함), 2 = 넘김·궤도 기지 소유 확인
sllm_probe_med_give = {
	set_variable = { which = sllm_med_q value = $SID$ }
	set_variable = { which = sllm_med_hit value = 0 }
	if = {
		limit = { has_country_flag = sllm_med_found }
		event_target:sllm_med_$FROM$ = {
			every_system_within_border = {
				limit = {
					is_variable_set = sllm_med_sid
					check_variable = { which = sllm_med_sid value = $SID$ }
					event_target:sllm_med_$TO$ = { has_claim = prev }
					OR = {
						starbase = { is_controlled_by = event_target:sllm_med_$TO$ }
						any_system_planet = { is_owned_by = event_target:sllm_med_$FROM$ is_controlled_by = event_target:sllm_med_$TO$ }
					}
				}
				save_event_target_as = sllm_med_sys
				event_target:sllm_med_player = { set_variable = { which = sllm_med_hit value = 1 } }
			}
		}
	}
	if = {
		limit = { check_variable = { which = sllm_med_hit value = 1 } }
		event_target:sllm_med_sys = {
			every_system_planet_colony = {
				limit = { is_owned_by = event_target:sllm_med_$FROM$ }
				set_owner = event_target:sllm_med_$TO$
			}
			starbase = { set_owner = event_target:sllm_med_$TO$ }
			if = {
				limit = { starbase = { is_owned_by = event_target:sllm_med_$TO$ } }
				event_target:sllm_med_player = { set_variable = { which = sllm_med_hit value = 2 } }
			}
		}
	}
	log = "SLLM:MEDGIVE/;/req=\\[This.sllm_med_req]/;/sid=\\[This.sllm_med_q]/;/hit=\\[This.sllm_med_hit]"
}

# 전쟁 종료(모든 참전국 이탈 → 전쟁 삭제, 바닐라 end_war_effect) + 두 당사국 휴전.
# peace 0 = 전쟁 못 찾음, 1 = 종전·휴전, 2 = 두 나라가 아직 (다른 전쟁으로) 교전 중
sllm_probe_med_peace = {
	set_variable = { which = sllm_med_peace value = 0 }
	if = {
		limit = { has_country_flag = sllm_med_found }
		event_target:sllm_med_war = { end_war_effect = yes }
		event_target:sllm_med_p1 = { set_truce = { target = event_target:sllm_med_p2 type = war } }
		event_target:sllm_med_p2 = { set_truce = { target = event_target:sllm_med_p1 type = war } }
		set_variable = { which = sllm_med_peace value = 1 }
		if = {
			limit = { event_target:sllm_med_p1 = { is_at_war_with = event_target:sllm_med_p2 } }
			set_variable = { which = sllm_med_peace value = 2 }
		}
	}
	log = "SLLM:MEDAPPLY_END/;/req=\\[This.sllm_med_req]/;/sid=\\[This.sllm_save_id]/;/found=\\[This.sllm_med_ok]/;/peace=\\[This.sllm_med_peace]"
	remove_country_flag = sllm_med_found
}

# SLLM Probe - F-42 (2026-10-15): 세이브를 불러오기 전에 표결이 시작된 안건은 on_resolution_put_to_vote 가 다시 오지 않는다.
# 불러올 때와 매월, 지금 표결 중인 안건이 있으면 RESOLUTION stage=vote resume=1 + RESKEY 를 다시 찍고, 없으면 stage=none 을 찍는다.
# 앱은 resume 줄을 이미 알고 있는 안건이면 무시하고(알림 한 번), stage=none 이면 남은 안건 정보를 비운다. 플레이어 국가 범위에서 부른다.
sllm_probe_gc_resume = {
	if = {
		limit = { is_voting_on_resolution = any }
		if = {
			limit = { is_galactic_community_member = yes }
			log = "SLLM:RESOLUTION/;/stage=vote/;/resume=1/;/member=1/;/date=\\[GetDate]/;/proposer=0/;/pname=/;/target=0/;/tname="
		}
		else = {
			log = "SLLM:RESOLUTION/;/stage=vote/;/resume=1/;/member=0/;/date=\\[GetDate]/;/proposer=0/;/pname=/;/target=0/;/tname="
		}
		sllm_probe_log_voting_key = yes
	}
	else = {
		log = "SLLM:RESOLUTION/;/stage=none/;/resume=1/;/date=\\[GetDate]"
	}
}

# SLLM Probe - 연방회의 '전쟁 선포 찬반' 안건이 통과한 뒤의 적용(2026-10-17). Stellaris 4.5.2 실제 동작은 게임 검수 필요.
# 플레이어 국가 범위에서 부른다: sllm_probe_fed_war = { TARGET = 대상국가번호(sllm_idx) }
# 연방 회장이 대상에게 선전포고한다(회원국은 연방 동맹으로 함께 참전). 결과는 SLLM:FEDWAR at=1(전쟁 중) / 0 으로 알린다.
# 로그 문자열에는 매개변수를 넣지 않는다(F-39). req 는 앱이 먼저 넣어 둔 sllm_fed_apply_req.
sllm_probe_fed_war = {
	set_variable = { which = sllm_fedwar_at value = 0 }
	federation = {
		every_member = {
			limit = { is_federation_leader = yes }
			save_event_target_as = sllm_fedwar_leader
		}
	}
	every_country = {
		limit = { is_variable_set = sllm_idx check_variable = { which = sllm_idx value = $TARGET$ } }
		save_event_target_as = sllm_fedwar_target
	}
	if = {
		limit = {
			exists = event_target:sllm_fedwar_leader
			exists = event_target:sllm_fedwar_target
			NOT = { event_target:sllm_fedwar_leader = { is_at_war_with = event_target:sllm_fedwar_target } }
		}
		event_target:sllm_fedwar_leader = { declare_war = { target = event_target:sllm_fedwar_target } }
	}
	if = {
		limit = {
			exists = event_target:sllm_fedwar_leader
			exists = event_target:sllm_fedwar_target
			event_target:sllm_fedwar_leader = { is_at_war_with = event_target:sllm_fedwar_target }
		}
		set_variable = { which = sllm_fedwar_at value = 1 }
	}
	log = "SLLM:FEDWAR/;/sid=\\[This.sllm_save_id]/;/req=\\[This.sllm_fed_apply_req]/;/at=\\[This.sllm_fedwar_at]"
}

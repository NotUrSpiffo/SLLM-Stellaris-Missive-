# SLLM Probe - 연설 버프 적용. 앱이 콘솔로 부른다(플레이어 국가 범위): sllm_probe_speech_war = { KEY = sllm_speech_war_2 DAYS = 720 N = 5 }
# 같은 종류의 이전 버프는 지우고 새로 건다(쌓이지 않고 갱신). 매 줄에 카운터 N 을 붙여 같은 로그 줄이 버려지지 않게 한다.

sllm_probe_speech_war = {
	remove_modifier = sllm_speech_war_1
	remove_modifier = sllm_speech_war_2
	remove_modifier = sllm_speech_war_3
	add_modifier = { modifier = $KEY$ days = $DAYS$ }
	if = { limit = { NOT = { is_variable_set = sllm_sp_a } } set_variable = { which = sllm_sp_a value = 0 } }
	change_variable = { which = sllm_sp_a value = 1 }
	set_variable = { which = sllm_sp_req value = $N$ }
	if = {
		limit = { has_modifier = sllm_speech_war_1 }
		log = "SLLM:SPEECH/;/phase=applied/;/key=sllm_speech_war_1/;/sid=\\[This.sllm_save_id]/;/req=\\[This.sllm_sp_req]/;/n=\\[This.sllm_sp_req]/;/a=\\[This.sllm_sp_a]"
	}
	else_if = {
		limit = { has_modifier = sllm_speech_war_2 }
		log = "SLLM:SPEECH/;/phase=applied/;/key=sllm_speech_war_2/;/sid=\\[This.sllm_save_id]/;/req=\\[This.sllm_sp_req]/;/n=\\[This.sllm_sp_req]/;/a=\\[This.sllm_sp_a]"
	}
	else_if = {
		limit = { has_modifier = sllm_speech_war_3 }
		log = "SLLM:SPEECH/;/phase=applied/;/key=sllm_speech_war_3/;/sid=\\[This.sllm_save_id]/;/req=\\[This.sllm_sp_req]/;/n=\\[This.sllm_sp_req]/;/a=\\[This.sllm_sp_a]"
	}
	else = {
		log = "SLLM:SPEECH/;/phase=failed/;/sid=\\[This.sllm_save_id]/;/req=\\[This.sllm_sp_req]/;/n=\\[This.sllm_sp_req]/;/a=\\[This.sllm_sp_a]"
	}
}

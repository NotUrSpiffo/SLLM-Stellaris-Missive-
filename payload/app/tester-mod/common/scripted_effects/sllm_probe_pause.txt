# SLLM Probe - report whether the game is paused (EXPERIMENT ONLY).
# The app runs "sllm_probe_pause_state = yes" in the player's scope through the console, then decides whether to press Space.
# Every line carries a counter, because the game drops identical repeated log lines.
sllm_probe_pause_state = {
	if = { limit = { NOT = { is_variable_set = sllm_ps_n } } set_variable = { which = sllm_ps_n value = 0 } }
	change_variable = { which = sllm_ps_n value = 1 }
	set_variable = { which = sllm_ps_p value = 0 }
	if = { limit = { is_paused = yes } set_variable = { which = sllm_ps_p value = 1 } }
	log = "SLLM:PSTATE/;/p=\\[This.sllm_ps_p]/;/n=\\[This.sllm_ps_n]"
}

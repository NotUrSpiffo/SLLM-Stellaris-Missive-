		export_trigger_value_to_variable = { trigger = defender_war_exhaustion variable = sllm_w_dx }

		}

	export_trigger_value_to_variable = { trigger = def_war_exhaustion_sum variable = sllm_cx_defex }

	set_variable = { which = sllm_cx_sys value = 0 }

		prev = { change_variable = { which = sllm_cx_sys value = 1 } }
		if = {
			limit = { prev = { has_claim = prev } }
			prev = { change_variable = { which = sllm_cx_claims value = 1 } }
		}
		if = {
			limit = { NOT = { is_surveyed = { who = prev status = yes } } }
			prev = { change_variable = { which = sllm_cx_unsv value = 1 } }
		}
	}
	# 월 수입(민간격려: 에너지·광물·식량·합금·소비재, 학구적: 연구 3종)과 행성 수

# SLLM allied council. Read who may sit, then apply one decision the game can already do.
# Eligibility: our subjects (every_subject), our overlord (has_any_overlord), or has_defensive_pact.
# Same-federation empires are left to the federation summit. Independence guarantees are not a seat.
# KIND 1 trade, 2 snub, 3 reaffirm trust+opinion, 4 open borders+trust+opinion,
# 5 declare_war with the mod casus belli, 6 leave a two-party war, 7 coordination modifier,
# 8 subject loyalty, 9 research aid. No effect here creates or cancels a diplomatic pact.

sllm_probe_ally_read = {
	every_playable_country = {
		limit = { is_ai = no }
		sllm_probe_init_vars = yes
		sllm_probe_assign_idx = yes
		set_variable = { which = sllm_ally_req value = $REQ$ }
		set_variable = { which = sllm_ally_n value = 0 }
		set_variable = { which = sllm_ally_ploy value = 0 }
		set_variable = { which = sllm_ally_phl value = 0 }
		if = {
			limit = { has_any_overlord = yes }
			export_trigger_value_to_variable = { trigger = has_loyalty variable = sllm_ally_ploy }
			set_variable = { which = sllm_ally_phl value = 1 }
		}
		log = "SLLM:ALLYBEGIN/;/req=\[This.sllm_ally_req]/;/sid=\[This.sllm_save_id]/;/n=0"
		every_subject = {
			limit = {
				is_variable_set = sllm_idx
				has_communications = prev
				NOT = { is_in_federation_with = prev }
			}
			set_variable = { which = sllm_ay_bond value = 1 }
			set_variable = { which = sllm_ally_req value = prev.sllm_ally_req }
			sllm_probe_ally_log = yes
			sllm_probe_ally_traits = yes
		}
		if = {
			limit = { has_any_overlord = yes }
			overlord = {
				if = {
					limit = {
						is_variable_set = sllm_idx
						has_communications = prev
						NOT = { is_in_federation_with = prev }
					}
					set_variable = { which = sllm_ay_bond value = 2 }
					set_variable = { which = sllm_ally_req value = prev.sllm_ally_req }
					sllm_probe_ally_log = yes
					sllm_probe_ally_traits = yes
				}
			}
		}
		every_country = {
			limit = {
				is_ai = yes
				is_variable_set = sllm_idx
				has_communications = prev
				has_defensive_pact = prev
				NOT = { is_same_value = prev }
				NOT = { is_in_federation_with = prev }
			}
			set_variable = { which = sllm_ay_bond value = 3 }
			set_variable = { which = sllm_ally_req value = prev.sllm_ally_req }
			sllm_probe_ally_log = yes
			sllm_probe_ally_traits = yes
		}
		log = "SLLM:ALLYEND/;/req=\[This.sllm_ally_req]/;/sid=\[This.sllm_save_id]/;/n=\[This.sllm_ally_n]"
	}
}

# Called from the country that should declare. Enemy was saved on this call only.
sllm_probe_ally_war_do = {
	if = {
		limit = { has_truce = event_target:sllm_ally_enemy }
		event_target:sllm_player = { set_variable = { which = sllm_ally_status value = 3 } }
	}
	else_if = {
		limit = { is_at_war_with = event_target:sllm_ally_enemy }
		event_target:sllm_player = { set_variable = { which = sllm_ally_status value = 4 } }
	}
	else = {
		if = {
			limit = { event_target:sllm_player = { check_variable = { which = sllm_in_reason value = 3 } } }
			add_casus_belli = { type = cb_sllm_humiliation who = event_target:sllm_ally_enemy days = 5 }
			declare_war = { target = event_target:sllm_ally_enemy attacker_war_goal = wg_sllm_humiliation }
		}
		else_if = {
			limit = { event_target:sllm_player = { check_variable = { which = sllm_in_reason value = 4 } } }
			add_casus_belli = { type = cb_sllm_tribute who = event_target:sllm_ally_enemy days = 5 }
			declare_war = { target = event_target:sllm_ally_enemy attacker_war_goal = wg_sllm_tribute }
		}
		else = {
			add_casus_belli = { type = cb_sllm_revenge who = event_target:sllm_ally_enemy days = 5 }
			declare_war = { target = event_target:sllm_ally_enemy attacker_war_goal = wg_sllm_revenge }
		}
		if = {
			limit = { NOT = { is_at_war_with = event_target:sllm_ally_enemy } }
			declare_war = { target = event_target:sllm_ally_enemy attacker_war_goal = wg_subjugation }
		}
		if = {
			limit = { NOT = { is_at_war_with = event_target:sllm_ally_enemy } }
			declare_war = { target = event_target:sllm_ally_enemy }
		}
		if = {
			limit = { is_at_war_with = event_target:sllm_ally_enemy }
			event_target:sllm_player = { set_variable = { which = sllm_ally_status value = 1 } set_variable = { which = sllm_last_result value = 1 } }
		}
		else = {
			event_target:sllm_player = { set_variable = { which = sllm_ally_status value = 10 } }
		}
	}
}

sllm_probe_ally_peace_do = {
	if = {
		limit = { NOT = { is_at_war_with = event_target:sllm_ally_enemy } }
		event_target:sllm_player = { set_variable = { which = sllm_ally_status value = 7 } }
	}
	else = {
		random_war = {
			limit = {
				is_war_participant = { who = prev }
				is_war_participant = { who = event_target:sllm_ally_enemy }
				count_war_participant = { count = 2 }
			}
			remove_war_participant = prev
		}
		if = {
			limit = { NOT = { is_at_war_with = event_target:sllm_ally_enemy } }
			event_target:sllm_player = { set_variable = { which = sllm_ally_status value = 1 } set_variable = { which = sllm_last_result value = 1 } }
		}
		else = {
			event_target:sllm_player = { set_variable = { which = sllm_ally_status value = 6 } }
		}
	}
}

sllm_probe_ally_coord_do = {
	if = {
		limit = { is_at_war_with = event_target:sllm_ally_enemy }
		remove_modifier = sllm_ally_coordination
		add_modifier = { modifier = sllm_ally_coordination days = 360 }
		event_target:sllm_player = { set_variable = { which = sllm_ally_status value = 1 } set_variable = { which = sllm_last_result value = 1 } }
	}
	else = {
		event_target:sllm_player = { set_variable = { which = sllm_ally_status value = 7 } }
	}
}

sllm_probe_ally_act = {
	every_playable_country = {
		limit = { is_ai = no }
		sllm_probe_init_vars = yes
		save_event_target_as = sllm_player
		set_variable = { which = sllm_in_req value = $REQ$ }
		set_variable = { which = sllm_in_kind value = $KIND$ }
		set_variable = { which = sllm_in_dir value = $DIR$ }
		set_variable = { which = sllm_in_amt value = $AMT$ }
		set_variable = { which = sllm_in_reason value = $REASON$ }
		set_variable = { which = sllm_in_actor value = $ACTOR$ }
		set_variable = { which = sllm_ally_status value = 5 }
		if = {
			limit = { NOT = { check_variable = { which = sllm_save_id value = $SID$ } } }
			set_variable = { which = sllm_ally_status value = 8 }
			log = "SLLM:ACK/;/req=$REQ$/;/ally/;/status=8/;/result=2"
		}
		else_if = {
			limit = { check_variable = { which = sllm_last_req value = $REQ$ } }
			log = "SLLM:ACK/;/seq=\[This.sllm_exec_seq]/;/req=\[This.sllm_in_req]/;/ally/;/dup/;/status=\[This.sllm_ally_status]/;/result=\[This.sllm_last_result]"
		}
		else_if = {
			limit = { check_variable = { which = sllm_last_req value > $REQ$ } }
			log = "SLLM:ACK/;/seq=\[This.sllm_exec_seq]/;/req=\[This.sllm_in_req]/;/ally/;/stale/;/status=\[This.sllm_ally_status]/;/result=\[This.sllm_last_result]/;/last=\[This.sllm_last_req]"
		}
		else = {
			set_variable = { which = sllm_last_req value = $REQ$ }
			change_variable = { which = sllm_exec_seq value = 1 }
			set_variable = { which = sllm_last_result value = 2 }
			if = {
				limit = { check_variable = { which = sllm_in_kind value = 1 } }
				random_country = {
					limit = { is_variable_set = sllm_idx check_variable = { which = sllm_idx value = $TID$ } }
					if = {
						limit = { event_target:sllm_player = { check_variable = { which = sllm_in_dir value = 1 } } }
						if = {
							limit = { event_target:sllm_player = { has_resource = { type = $RES$ amount >= $AMT$ } } }
							event_target:sllm_player = { add_resource = { $RES$ = -$AMT$ } }
							add_resource = { $RES$ = $AMT$ }
							event_target:sllm_player = { set_variable = { which = sllm_ally_status value = 1 } set_variable = { which = sllm_last_result value = 1 } }
						}
						else = { event_target:sllm_player = { set_variable = { which = sllm_ally_status value = 2 } } }
					}
					else = {
						if = {
							limit = { has_resource = { type = $RES$ amount >= $AMT$ } }
							add_resource = { $RES$ = -$AMT$ }
							event_target:sllm_player = { add_resource = { $RES$ = $AMT$ } set_variable = { which = sllm_ally_status value = 1 } set_variable = { which = sllm_last_result value = 1 } }
						}
						else = { event_target:sllm_player = { set_variable = { which = sllm_ally_status value = 2 } } }
					}
				}
			}
			else_if = {
				limit = { check_variable = { which = sllm_in_kind value = 2 } }
				random_country = {
					limit = { is_variable_set = sllm_idx check_variable = { which = sllm_idx value = $TID$ } }

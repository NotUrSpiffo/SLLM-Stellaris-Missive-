# 전쟁이 시작됐다 (Root = 국가, From = 전쟁)
country_event = {
	id = sllm_radio.11
	hide_window = yes
	is_triggered_only = yes
	immediate = {
		if = {
			limit = { is_ai = no }
			sllm_probe_evt_bump = yes
			log = "SLLM:EVT/;/kind=war_start/;/date=\\[GetDate]/;/n=\\[This.sllm_ev_n]/;/who=\\[This.GetName]/;/war=\\[From.GetName]"
			# 전쟁 정보(명분·주도국·관계) — 2026-10-20 알파 10. 규칙: common/scripted_effects/sllm_probe_warinfo.txt
			save_event_target_as = sllm_wi_player
			from = { sllm_probe_warinfo = yes sllm_probe_warvoice = yes }
		}
		# 남의 전쟁(우리가 참전하지 않은 전쟁) — 2026-10-20 알파 10 규탄 성명 방향 2. 주 공격국 범위에서 한 번만, 플레이어가 양쪽 중 하나와 통신할 때만.
		else_if = {
			limit = {
				is_ai = yes
				from = { attacker = { is_same_value = root } }
				any_playable_country = { is_ai = no OR = { has_communications = root from = { defender = { has_communications = prevprev } } } }
			}
			random_playable_country = { limit = { is_ai = no } save_event_target_as = sllm_wi_player }
			if = {
				limit = { from = { NOT = { is_war_participant = { who = event_target:sllm_wi_player side = attackers } } NOT = { is_war_participant = { who = event_target:sllm_wi_player side = defenders } } } }
				from = { sllm_probe_foreignwar = yes }
			}
		}
	}
}
